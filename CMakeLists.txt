# dbosoft Windows build for OVN — CMake + MSVC, no MSYS/autotools/cccl.
#
# Clean OVERLAY build, mirroring the OVS overlay (ovs/CMakeLists.txt).  It exists
# because upstream OVS removed Windows support: OVN's autotools/MSYS build consumes
# an *autotools-built* OVS tree (OVS_BUILDDIR/lib + OVS Windows shim header dirs).
# Once OVS moves to the CMake overlay there is no autotools OVS build dir / .la to
# link, so OVN's MSYS build can no longer be wired up.  This overlay builds OVN's
# userspace against the OVS CMake build instead, MSYS-free.
#
# Builds libovn (the OVN core library) against the OVS CMake libraries, with OVN's
# OVSDB IDL (ovn-nb/sb/ic-nb/ic-sb/br) generated here via OVS's ovsdb-idlc.
cmake_minimum_required(VERSION 3.20)
project(ovn_windows C)

if(NOT MSVC)
  message(FATAL_ERROR "The dbosoft Windows OVN CMake build targets MSVC (cl).")
endif()

set(OVN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(OVS_ROOT "${OVN_ROOT}/ovs")
set(PTHREADS_ROOT "C:/PTHREADS-BUILT" CACHE PATH "pthreads4w install prefix")
set(OPENSSL_ROOT  "C:/OpenSSL-Win64"  CACHE PATH "OpenSSL install prefix")
set(CMAKE_C_STANDARD 11)

# --- Build the OVS submodule via its own CMake overlay ----------------------
# Exposes the targets openvswitch / ovsdb / ofproto and, at CONFIGURE time,
# generates all OVS headers (config.h, version.h, the string.h/stdio.h MSVC
# shims, odp-netlink*.h, IDL) into ${OVS_GEN}.  OVN compiles against those.
add_subdirectory(ovs)
set(OVS_GEN "${CMAKE_BINARY_DIR}/gen")   # ovs/CMakeLists.txt uses ${CMAKE_BINARY_DIR}/gen

# OVN-generated files go to a SEPARATE dir (${GEN}) listed FIRST on the include
# path so OVN's config.h (this version) wins over OVS's config.h in ${OVS_GEN}.
# ${OVS_GEN} itself must still be on the path (after ${GEN}) because OVN includes
# OVS's generated IDL headers as "lib/vswitch-idl.h", which only resolves with the
# OVS gen root on the path -- not just its include/ and lib/ subdirs.
set(GEN "${CMAKE_BINARY_DIR}/gen-ovn")
file(MAKE_DIRECTORY "${GEN}" "${GEN}/lib" "${GEN}/include/ovn")

# Derive the OVN version from configure.ac (AC_INIT) so it tracks each resync
# instead of going stale.
file(READ "${OVN_ROOT}/configure.ac" _ovn_configure_ac)
string(REGEX MATCH "AC_INIT\\(ovn, ([0-9]+\\.[0-9]+\\.[0-9]+)" _ovn_ac_match "${_ovn_configure_ac}")
set(OVN_VERSION "${CMAKE_MATCH_1}")
if(NOT OVN_VERSION)
  message(FATAL_ERROR "Could not derive OVN_VERSION from configure.ac AC_INIT")
endif()
set(VERSION_SUFFIX "")
configure_file("${OVN_ROOT}/cmake/config.h.in" "${GEN}/config.h" @ONLY)

# include/ovn/version.h
set(PACKAGE_STRING  "ovn ${OVN_VERSION}")
set(PACKAGE_VERSION "${OVN_VERSION}")
set(LT_CURRENT 0)
set(LT_REVISION 0)
set(LT_AGE 0)
configure_file("${OVN_ROOT}/include/ovn/version.h.in"
               "${GEN}/include/ovn/version.h" @ONLY)

# lib/ovn-dirs.c — the @VAR@ substitutions must include the surrounding quotes
# (the autotools sed wraps each value in ""); srcdir stays bare (used in #line).
set(srcdir     "${OVN_ROOT}")
set(LOGDIR     "\"C:/ProgramData/openvswitch/var/log/ovn\"")
set(OVN_RUNDIR "\"C:/ProgramData/openvswitch/var/run/ovn\"")
set(DBDIR      "\"C:/ProgramData/openvswitch/etc/ovn\"")
set(bindir     "\"C:/openvswitch/usr/bin\"")
set(sysconfdir "\"C:/ProgramData/openvswitch/etc\"")
set(pkgdatadir "\"C:/openvswitch/usr/share/ovn\"")
configure_file("${OVN_ROOT}/lib/ovn-dirs.c.in" "${GEN}/lib/ovn-dirs.c" @ONLY)

# --- OVN IDL codegen (5 schemas) via OVS's ovsdb-idlc -----------------------
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ENV{PYTHONPATH} "${OVS_ROOT}/python")   # ovsdb-idlc imports ovs.* (incl. ovs.dirs, generated by the OVS subdir)
set(IDLC "${OVS_ROOT}/ovsdb/ovsdb-idlc.in")

function(ovn_gen out)
  execute_process(
    COMMAND "${Python3_EXECUTABLE}" ${ARGN}
    OUTPUT_FILE "${out}"
    WORKING_DIRECTORY "${OVN_ROOT}"
    RESULT_VARIABLE _rc)
  if(NOT _rc EQUAL 0)
    message(FATAL_ERROR "OVN codegen failed (rc=${_rc}) for ${out}\n  ${Python3_EXECUTABLE} ${ARGN}")
  endif()
endfunction()

# (schema-base, schema-file) — annotate(schema+ann) -> .ovsidl -> c-idl-source/header
set(OVN_IDLS
  "ovn-nb-idl|ovn-nb.ovsschema"
  "ovn-sb-idl|ovn-sb.ovsschema"
  "ovn-ic-nb-idl|ovn-ic-nb.ovsschema"
  "ovn-ic-sb-idl|ovn-ic-sb.ovsschema"
  "ovn-br-idl|ovn-br.ovsschema")
set(OVN_IDL_GEN_SOURCES "")
foreach(_idl IN LISTS OVN_IDLS)
  string(REPLACE "|" ";" _parts "${_idl}")
  list(GET _parts 0 _base)
  list(GET _parts 1 _schema)
  ovn_gen("${GEN}/lib/${_base}.ovsidl" "${IDLC}" annotate
          "${OVN_ROOT}/${_schema}" "${OVN_ROOT}/lib/${_base}.ann")
  ovn_gen("${GEN}/lib/${_base}.c" "${IDLC}" c-idl-source "${GEN}/lib/${_base}.ovsidl")
  ovn_gen("${GEN}/lib/${_base}.h" "${IDLC}" c-idl-header "${GEN}/lib/${_base}.ovsidl")
  list(APPEND OVN_IDL_GEN_SOURCES "${GEN}/lib/${_base}.c")
endforeach()

# --- libovn -----------------------------------------------------------------
set(OVN_LIB_SOURCES
  lib/acl-log.c lib/actions.c lib/chassis-index.c lib/copp.c lib/expr.c
  lib/extend-table.c lib/features.c lib/ovn-parallel-hmap.c lib/ip-mcast-index.c
  lib/lflow-conj-ids.c lib/mac-binding-index.c lib/mcast-group-index.c
  lib/memory-trim.c lib/lex.c lib/objdep.c lib/ofctrl-seqno.c lib/ovn-l7.c
  lib/ovn-util.c lib/logical-fields.c lib/inc-proc-eng.c lib/lb.c
  lib/sparse-array.c lib/vec.c lib/vif-plug-provider.c
  lib/vif-plug-providers/dummy/vif-plug-dummy.c)

add_library(ovn STATIC
  ${OVN_LIB_SOURCES}
  "${GEN}/lib/ovn-dirs.c"
  ${OVN_IDL_GEN_SOURCES})

# Include order: OVN generated (config.h/idl/version.h) FIRST so OVN's config.h
# wins; then OVN source; then OVS source + OVS generated (include/ + lib/ shims).
set(OVN_INCLUDES
  "${GEN}" "${GEN}/lib" "${GEN}/include"
  "${OVN_ROOT}" "${OVN_ROOT}/include" "${OVN_ROOT}/lib" "${OVN_ROOT}/ovn"
  "${OVN_ROOT}/include/windows"
  "${OVS_ROOT}" "${OVS_ROOT}/include" "${OVS_ROOT}/include/windows"
  "${OVS_ROOT}/lib" "${OVS_ROOT}/datapath-windows/include"
  "${OVS_GEN}" "${OVS_GEN}/include" "${OVS_GEN}/lib"
  "${PTHREADS_ROOT}/include" "${OPENSSL_ROOT}/include")
set(OVN_DEFS HAVE_CONFIG_H _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS
             _WINSOCK_DEPRECATED_NO_WARNINGS)

# /Zc:inline- is load-bearing: OVN, like OVS, relies on OVS_CONSTRUCTOR/.CRT$XCU
# self-registration (VLOG_DEFINE_THIS_MODULE etc.); the VS-generator Release
# default /Zc:inline strips those unreferenced CRT initializers at compile time.
function(ovn_setup t)
  target_include_directories(${t} PRIVATE ${OVN_INCLUDES})
  target_compile_definitions(${t} PRIVATE ${OVN_DEFS})
  target_compile_options(${t} PRIVATE /W1 /Zc:inline-)
endfunction()
ovn_setup(ovn)

# --- OVN executables --------------------------------------------------------
# The daemons (ovn-northd, ovn-controller) and the *ctl/appctl/trace utilities,
# linked against libovn + the OVS CMake libraries.  These are the userspace
# binaries the install/package step ships (the autotools build's bin_PROGRAMS).
set(OVN_EXT_LIBDIRS "${OPENSSL_ROOT}/lib" "${OPENSSL_ROOT}/lib/VC/x64/MT"
                    "${PTHREADS_ROOT}/lib")
set(OVN_SYS_LIBS ws2_32 Shlwapi iphlpapi wbemuuid ole32 oleaut32)

function(ovn_exe t)
  ovn_setup(${t})
  target_link_directories(${t} PRIVATE ${OVN_EXT_LIBDIRS})
  # ovn + ovsdb + openvswitch covers every binary's LDADD (controller needs only
  # ovn + openvswitch; the extra static ovsdb pulls nothing unreferenced).
  target_link_libraries(${t} PRIVATE
    ovn ovsdb openvswitch libssl libcrypto pthreadVC3 ${OVN_SYS_LIBS})
endfunction()

# ovn-controller: Windows has no netlink, so build the *-stub.c variants and drop
# the HAVE_NETLINK sources (mirrors the automake `else` branch).  Controller test
# programs (test-*.c) are not binaries we ship.
file(GLOB OVN_CONTROLLER_SRC CONFIGURE_DEPENDS "${OVN_ROOT}/controller/*.c")
list(FILTER OVN_CONTROLLER_SRC EXCLUDE REGEX
  "/(test-lflow-cache|test-vif-plug|host-if-monitor|ovn-netlink-notifier|neighbor-exchange-netlink|neighbor-exchange|nexthop-exchange|route-exchange-netlink|route-exchange)\\.c$")
add_executable(ovn-controller ${OVN_CONTROLLER_SRC})
ovn_exe(ovn-controller)

# Same controller sources minus main(), reused by ovntest so the controller
# test drivers (test-lflow-cache, test-vif-plug) link against the identical
# object set that ovn-controller links (guaranteed symbol resolution).
set(OVN_CONTROLLER_TESTOBJ ${OVN_CONTROLLER_SRC})
list(FILTER OVN_CONTROLLER_TESTOBJ EXCLUDE REGEX "/ovn-controller\\.c$")

# ovn-northd: all of northd/ is portable; only the test driver is excluded.
file(GLOB OVN_NORTHD_SRC CONFIGURE_DEPENDS "${OVN_ROOT}/northd/*.c")
list(FILTER OVN_NORTHD_SRC EXCLUDE REGEX "/test-ipam\\.c$")
add_executable(ovn-northd ${OVN_NORTHD_SRC})
ovn_exe(ovn-northd)

add_executable(ovn-nbctl utilities/ovn-dbctl.c utilities/ovn-nbctl.c)
ovn_exe(ovn-nbctl)
add_executable(ovn-sbctl utilities/ovn-dbctl.c utilities/ovn-sbctl.c)
ovn_exe(ovn-sbctl)
add_executable(ovn-trace utilities/ovn-trace.c)
ovn_exe(ovn-trace)
add_executable(ovn-appctl utilities/ovn-appctl.c)
ovn_exe(ovn-appctl)

# --- install (the C:/openvswitch/usr layout the packaging scripts consume) ---
# Run with: cmake --install <build> --config Release --prefix C:/openvswitch/usr
# Mirrors the autotools "make install" split: daemons -> sbin, tools -> bin,
# schemas -> share/{openvswitch,ovn}.  Both OVN and the OVS subproject targets
# are installed from here (cross-directory install(TARGETS), CMake >= 3.13) so the
# OVS submodule overlay stays a pure build definition with no install rules.
install(TARGETS ovn-controller ovn-northd ovn-nbctl ovn-sbctl ovn-trace ovn-appctl
        RUNTIME DESTINATION bin)
install(TARGETS ovs-vswitchd ovsdb-server RUNTIME DESTINATION sbin)
install(TARGETS ovs-vsctl ovs-dpctl ovs-ofctl ovs-appctl ovsdb-tool ovsdb-client
        RUNTIME DESTINATION bin)
install(FILES "${OVN_ROOT}/ovn-nb.ovsschema" "${OVN_ROOT}/ovn-sb.ovsschema"
        DESTINATION share/ovn)
install(FILES "${OVS_ROOT}/vswitchd/vswitch.ovsschema"
              "${OVS_ROOT}/ovsdb/local-config.ovsschema"
        DESTINATION share/openvswitch)

# --- tests (CMake/CTest unit-test net; MSYS-free, no autotools "make check") --
# Reuse OVN's C unit-test modules the way the OVS overlay does: compile the
# OVSTEST_REGISTER modules that link against libovn into one multi-call binary
# and drive them through CTest.  Named `ovntest` (not `ovstest`) to avoid
# clashing with the OVS subproject's own ovstest target.  The autotools autotest
# suite (tests/*.at -> tests/testsuite) is not used on Windows.
#
# libovn-level modules plus the controller/northd test drivers, which link the
# same controller sources as ovn-controller (OVN_CONTROLLER_TESTOBJ) plus
# northd/ipam.c.  The netlink test (tests/test-ovn-netlink.c) stays out
# (HAVE_NETLINK/Linux only).
set(OVNTEST_SOURCES
  tests/ovstest.c
  tests/test-utils.c
  tests/test-ovn.c
  tests/test-sparse-array.c
  tests/test-vector.c
  lib/test-lflow-conj-ids.c
  lib/test-ovn-features.c
  lib/test-ofctrl-seqno.c
  controller/test-lflow-cache.c
  controller/test-vif-plug.c
  northd/test-ipam.c
  northd/ipam.c)
add_executable(ovntest ${OVNTEST_SOURCES} ${OVN_CONTROLLER_TESTOBJ})
ovn_exe(ovntest)

# Configure-time drift guard: FATAL if an upstream-added tests/test-*.c registers
# an OVSTEST module but is missing from OVNTEST_SOURCES.
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/check-ovntest-list.cmake)

enable_testing()
set(_ovntest_dll_path
  "PATH=path_list_prepend:${PTHREADS_ROOT}/bin;PATH=path_list_prepend:${OPENSSL_ROOT}")

# Self-contained modules (run all their own asserts; exit code is the verdict).
set(OVNTEST_EXIT_CASES
  "test-vector add"
  "test-sparse-array add"
  "test-ovn-features run")
foreach(_case IN LISTS OVNTEST_EXIT_CASES)
  separate_arguments(_argv UNIX_COMMAND "${_case}")
  string(REPLACE " " "_" _tname "ovntest_${_case}")
  add_test(NAME ${_tname} COMMAND ovntest ${_argv})
  set_tests_properties(${_tname} PROPERTIES ENVIRONMENT_MODIFICATION "${_ovntest_dll_path}")
endforeach()

# /Zc:inline- registration canary: a stripped OVS_CONSTRUCTOR => empty module list.
add_test(NAME ovntest-help COMMAND ovntest --help)
set_tests_properties(ovntest-help PROPERTIES
  PASS_REGULAR_EXPRESSION "test-ovn"
  ENVIRONMENT_MODIFICATION "${_ovntest_dll_path}")
