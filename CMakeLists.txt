# dbosoft Windows build for OVN — CMake + MSVC, no MSYS/autotools/cccl.
#
# Clean OVERLAY build, mirroring the OVS overlay (ovs/CMakeLists.txt).  It exists
# because upstream OVS removed Windows support: OVN's autotools/MSYS build consumes
# an *autotools-built* OVS tree (OVS_BUILDDIR/lib + OVS Windows shim header dirs).
# Once OVS moves to the CMake overlay there is no autotools OVS build dir / .la to
# link, so OVN's MSYS build can no longer be wired up.  This overlay builds OVN's
# userspace against the OVS CMake build instead, MSYS-free.
#
# Builds libovn (the OVN core library) against the OVS CMake libraries, with OVN's
# OVSDB IDL (ovn-nb/sb/ic-nb/ic-sb/br) generated here via OVS's ovsdb-idlc.
cmake_minimum_required(VERSION 3.20)
project(ovn_windows C)

if(NOT MSVC)
  message(FATAL_ERROR "The dbosoft Windows OVN CMake build targets MSVC (cl).")
endif()

set(OVN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(OVS_ROOT "${OVN_ROOT}/ovs")
set(PTHREADS_ROOT "C:/PTHREADS-BUILT" CACHE PATH "pthreads4w install prefix")
set(OPENSSL_ROOT  "C:/OpenSSL-Win64"  CACHE PATH "OpenSSL install prefix")
set(CMAKE_C_STANDARD 11)

# --- Build the OVS submodule via its own CMake overlay ----------------------
# Exposes the targets openvswitch / ovsdb / ofproto and, at CONFIGURE time,
# generates all OVS headers (config.h, version.h, the string.h/stdio.h MSVC
# shims, odp-netlink*.h, IDL) into ${OVS_GEN}.  OVN compiles against those.
add_subdirectory(ovs)
set(OVS_GEN "${CMAKE_BINARY_DIR}/gen")   # ovs/CMakeLists.txt uses ${CMAKE_BINARY_DIR}/gen

# OVN-generated files go to a SEPARATE dir so OVN's config.h (this version) wins
# over OVS's config.h; ${OVS_GEN} itself is intentionally NOT on OVN's include
# path (only its include/ and lib/ subdirs), so OVN never picks up OVS's config.h.
set(GEN "${CMAKE_BINARY_DIR}/gen-ovn")
file(MAKE_DIRECTORY "${GEN}" "${GEN}/lib" "${GEN}/include/ovn")

set(OVN_VERSION "26.03.1")   # TODO: derive from configure.ac AC_INIT on resync
set(VERSION_SUFFIX "")
configure_file("${OVN_ROOT}/cmake/config.h.in" "${GEN}/config.h" @ONLY)

# include/ovn/version.h
set(PACKAGE_STRING  "ovn ${OVN_VERSION}")
set(PACKAGE_VERSION "${OVN_VERSION}")
set(LT_CURRENT 0)
set(LT_REVISION 0)
set(LT_AGE 0)
configure_file("${OVN_ROOT}/include/ovn/version.h.in"
               "${GEN}/include/ovn/version.h" @ONLY)

# lib/ovn-dirs.c — the @VAR@ substitutions must include the surrounding quotes
# (the autotools sed wraps each value in ""); srcdir stays bare (used in #line).
set(srcdir     "${OVN_ROOT}")
set(LOGDIR     "\"C:/ProgramData/ovn/var/log/ovn\"")
set(OVN_RUNDIR "\"C:/ProgramData/ovn/var/run/ovn\"")
set(DBDIR      "\"C:/ProgramData/ovn/etc/ovn\"")
set(bindir     "\"C:/ovn/usr/bin\"")
set(sysconfdir "\"C:/ProgramData/ovn/etc\"")
set(pkgdatadir "\"C:/ovn/usr/share/ovn\"")
configure_file("${OVN_ROOT}/lib/ovn-dirs.c.in" "${GEN}/lib/ovn-dirs.c" @ONLY)

# --- OVN IDL codegen (5 schemas) via OVS's ovsdb-idlc -----------------------
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ENV{PYTHONPATH} "${OVS_ROOT}/python")   # ovsdb-idlc imports ovs.* (incl. ovs.dirs, generated by the OVS subdir)
set(IDLC "${OVS_ROOT}/ovsdb/ovsdb-idlc.in")

function(ovn_gen out)
  execute_process(
    COMMAND "${Python3_EXECUTABLE}" ${ARGN}
    OUTPUT_FILE "${out}"
    WORKING_DIRECTORY "${OVN_ROOT}"
    RESULT_VARIABLE _rc)
  if(NOT _rc EQUAL 0)
    message(FATAL_ERROR "OVN codegen failed (rc=${_rc}) for ${out}\n  ${Python3_EXECUTABLE} ${ARGN}")
  endif()
endfunction()

# (schema-base, schema-file) — annotate(schema+ann) -> .ovsidl -> c-idl-source/header
set(OVN_IDLS
  "ovn-nb-idl|ovn-nb.ovsschema"
  "ovn-sb-idl|ovn-sb.ovsschema"
  "ovn-ic-nb-idl|ovn-ic-nb.ovsschema"
  "ovn-ic-sb-idl|ovn-ic-sb.ovsschema"
  "ovn-br-idl|ovn-br.ovsschema")
set(OVN_IDL_GEN_SOURCES "")
foreach(_idl IN LISTS OVN_IDLS)
  string(REPLACE "|" ";" _parts "${_idl}")
  list(GET _parts 0 _base)
  list(GET _parts 1 _schema)
  ovn_gen("${GEN}/lib/${_base}.ovsidl" "${IDLC}" annotate
          "${OVN_ROOT}/${_schema}" "${OVN_ROOT}/lib/${_base}.ann")
  ovn_gen("${GEN}/lib/${_base}.c" "${IDLC}" c-idl-source "${GEN}/lib/${_base}.ovsidl")
  ovn_gen("${GEN}/lib/${_base}.h" "${IDLC}" c-idl-header "${GEN}/lib/${_base}.ovsidl")
  list(APPEND OVN_IDL_GEN_SOURCES "${GEN}/lib/${_base}.c")
endforeach()

add_library(ovn STATIC
  ${OVN_LIB_SOURCES}
  "${GEN}/lib/ovn-dirs.c"
  ${OVN_IDL_GEN_SOURCES})

# Include order: OVN generated (config.h/idl/version.h) FIRST so OVN's config.h
# wins; then OVN source; then OVS source + OVS generated (include/ + lib/ shims).
set(OVN_INCLUDES
  "${GEN}" "${GEN}/lib" "${GEN}/include"
  "${OVN_ROOT}" "${OVN_ROOT}/include" "${OVN_ROOT}/lib" "${OVN_ROOT}/ovn"
  "${OVN_ROOT}/include/windows"
  "${OVS_ROOT}" "${OVS_ROOT}/include" "${OVS_ROOT}/include/windows"
  "${OVS_ROOT}/lib" "${OVS_ROOT}/datapath-windows/include"
  "${OVS_GEN}/include" "${OVS_GEN}/lib"
  "${PTHREADS_ROOT}/include" "${OPENSSL_ROOT}/include")
set(OVN_DEFS HAVE_CONFIG_H _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS
             _WINSOCK_DEPRECATED_NO_WARNINGS)

# /Zc:inline- is load-bearing: OVN, like OVS, relies on OVS_CONSTRUCTOR/.CRT$XCU
# self-registration (VLOG_DEFINE_THIS_MODULE etc.); the VS-generator Release
# default /Zc:inline strips those unreferenced CRT initializers at compile time.
function(ovn_setup t)
  target_include_directories(${t} PRIVATE ${OVN_INCLUDES})
  target_compile_definitions(${t} PRIVATE ${OVN_DEFS})
  target_compile_options(${t} PRIVATE /W1 /Zc:inline-)
endfunction()
ovn_setup(ovn)
